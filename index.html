function atualizarAberturaPersiana(val) { percentualAbertura = parseFloat(val); const progresso = percentualAbertura / 100; document.getElementById('txt-abertura-val').innerText = `${Math.round(progresso * 100)}%`; const modeloKey = document.getElementById('select-modelo').value; if (modeloKey === 'double_vision') { // Double Vision corrigido: traspasse contínuo subindo em direção ao topo tecidoFrontMesh.visible = true; tecidoBackMesh.visible = true; if (grupoGomos) grupoGomos.visible = false; if (grupoLâminasVerticais) grupoLâminasVerticais.visible = false; if (pesoMesh) pesoMesh.visible = true; const alturaBase = 1.8; const fatorRolagem = Math.max(0.02, 1 - progresso); // Ambas as camadas sobem em conjunto realizando o traspasse das faixas tecidoFrontMesh.scale.y = fatorRolagem; tecidoFrontMesh.position.y = (progresso * (alturaBase / 2)); tecidoBackMesh.scale.y = fatorRolagem; tecidoBackMesh.position.y = (progresso * (alturaBase / 2)); pesoMesh.position.y = -0.9 + (progresso * 1.75); } else if (modeloKey === 'romana') { tecidoFrontMesh.visible = false; tecidoBackMesh.visible = false; if (grupoLâminasVerticais) grupoLâminasVerticais.visible = false; if (grupoGomos) { grupoGomos.visible = true; const gomos = grupoGomos.children; const totalGomos = gomos.length; // Limitado a 4 gomos conforme solicitado const gomosRecolhidos = Math.floor(progresso * totalGomos); gomos.forEach((gomo, index) => { if (index < gomosRecolhidos) { // Gomos recolhidos e empilhados ordenadamente em cima gomo.position.y = 0.82 - (index * 0.04); gomo.scale.y = 0.25; gomo.visible = true; } else { // Gomos pendurados esticados para baixo gomo.position.y = 0.6 - (index * 0.35); gomo.scale.y = 1; gomo.visible = (progresso < 0.98); } }); } if (pesoMesh) pesoMesh.visible = false; } else if (modeloKey === 'vertical') { // (Mantém a lógica da Vertical já ajustada anteriormente) tecidoFrontMesh.visible = false; tecidoBackMesh.visible = false; if (grupoGomos) grupoGomos.visible = false; if (pesoMesh) pesoMesh.visible = false; if (grupoLâminasVerticais) { grupoLâminasVerticais.visible = true; const tipoAbertura = document.querySelector('input[name="tipo-abertura-vert"]:checked').value; const lâminas = grupoLâminasVerticais.children; const total = lâminas.length; const larguraJanela = 1.48; const larguraLaminaOriginal = larguraJanela / total; lâminas.forEach((lam, index) => { const posXOriginal = -0.74 + (index * larguraLaminaOriginal) + (larguraLaminaOriginal / 2); if (tipoAbertura === 'lateral') { const fatorEmpilhamento = 1 - progresso; lam.position.x = -0.74 + (posXOriginal - (-0.74)) * (1 - progresso) + (progresso * 0.02); lam.scale.x = Math.max(0.04, fatorEmpilhamento); lam.visible = true; } else { const meio = total / 2; const distDoCentro = index - meio; if (progresso === 0) { lam.position.x = posXOriginal; lam.scale.x = 1; lam.visible = true; } else { const direcao = distDoCentro >= 0 ? 1 : -1; const alvoX = direcao > 0 ? (0.74 - Math.abs(distDoCentro) * 0.02) : (-0.74 + Math.abs(distDoCentro) * 0.02); lam.position.x = posXOriginal + (alvoX - posXOriginal) * progresso; lam.scale.x = Math.max(0.04, 1 - progresso); lam.visible = true; } } }); } } else { // Rolô Comum tecidoFrontMesh.visible = true; tecidoBackMesh.visible = false; if (grupoGomos) grupoGomos.visible = false; if (grupoLâminasVerticais) grupoLâminasVerticais.visible = false; if (pesoMesh) pesoMesh.visible = true; const alturaBase = 1.8; const porcentagemFechada = (100 - percentualAbertura) / 100; const novaAlturaGeo = Math.max(0.05, alturaBase * porcentagemFechada); tecidoFrontMesh.scale.y = Math.max(0.02, porcentagemFechada); tecidoFrontMesh.position.y = 0.9 - (novaAlturaGeo / 2); if (pesoMesh) pesoMesh.position.y = 0.9 - novaAlturaGeo; } }
